// BoardVerse.API/Controllers/StaffScheduleController.cs
private async Task EnsureCafeManagerAsync(Guid cafeId, CancellationToken ct)
{
    var callerId = _currentUser.GetUserId();
    var isAdmin = _currentUser.IsInRole("Admin");
    if (isAdmin) return; // bypass

}

### 3. Max duration 16h validation (M3)

```csharp
// BoardVerse.Services/Services/StaffScheduleService.cs
private const int MaxShiftMinutes = 16 * 60; // 960 phút

private TimeSpan CalculateShiftDuration(TimeOnly startTime, TimeOnly endTime)
{
    if (endTime <= startTime)
        return TimeSpan.FromHours(24) - (startTime - endTime); // overnight wrap
    return endTime - startTime;
}

private void ValidateShiftDuration(TimeOnly startTime, TimeOnly endTime)
{
    var duration = CalculateShiftDuration(startTime, endTime);
    if (duration.TotalMinutes > MaxShiftMinutes)
        throw new BadRequestException(ApiErrorMessages.StaffSchedule.DurationExceeds16Hours(duration.TotalHours));
}
```

### 4. Atomic bulk create (H1)

```csharp
// BoardVerse.Services/Services/StaffScheduleService.cs
public async Task<List<StaffScheduleResponseDto>> BulkCreateAsync(
    Guid cafeId, BulkCreateStaffSchedulesRequest request, Guid callerId, CancellationToken ct)
{
    var ownedTx = await _db.Database.BeginTransactionAsync(IsolationLevel.Serializable, ct);
    try
    {
        // 1. Validate ownership + max duration cho TẤT CẢ schedules trước
        foreach (var schedule in request.Schedules)
        {
            ValidateShiftDuration(schedule.StartTime, schedule.EndTime);
            await EnsureNoOverlapAsync(cafeId, schedule, ct);
        }

        // 2. Insert tất cả (giờ chỉ insert nếu mọi validate pass)
        var entities = new List<StaffSchedule>();
        foreach (var schedule in request.Schedules)
        {
            entities.Add(MapToEntity(cafeId, schedule));
        }
        await _repository.AddRangeAsync(entities, ct);
        await _db.SaveChangesAsync(ct);

        // 3. Commit 1 lần duy nhất
        await ownedTx.CommitAsync(ct);
        return entities.Select(MapToDto).ToList();
    }
    catch
    {
        await ownedTx.RollbackAsync(ct);
        throw;
    }
    finally
    {
        await ownedTx.DisposeAsync();
    }
}
```

### 6. Staff delete cancel schedules (C6)

```csharp
// BoardVerse.Services/Services/StaffScheduleService.cs
public async Task CancelAllSchedulesForStaffAsync(
    Guid cafeId, Guid staffUserId, CancellationToken ct)
{
    var ownedTx = await _db.Database.BeginTransactionAsync(IsolationLevel.Serializable, ct);
    try
    {
        var activeSchedules = await _repository.GetActiveSchedulesAsync(
            cafeId, staffUserId, ct);

        foreach (var schedule in activeSchedules)
        {
            schedule.Status = StaffScheduleStatus.Cancelled;
            schedule.UpdatedAt = DateTime.UtcNow;
        }

        await _db.SaveChangesAsync(ct);
        await ownedTx.CommitAsync(ct);

        _logger.LogInformation(
            "Cancelled {Count} schedules for staff {StaffId} removed from cafe {CafeId}",
            activeSchedules.Count, staffUserId, cafeId);
    }
    catch
    {
        await ownedTx.RollbackAsync(ct);
        throw;
    }
    finally
    {
        await ownedTx.DisposeAsync();
    }
}
```

## Validation chain sau khi fix

`POST /api/v1/cafes/{cafeId}/staff-schedules` validate theo thứ tự:

1. **Ownership** (C5, IDOR-01) — `cafe.ManagerId == callerId` (Admin bypass). → 403 `CrossCafeAccessDenied`.
2. **Staff là thành viên cafe** — `staffUserId ∈ CafeStaffs` của cafe. → 400 `NotCafeStaff`.
3. **StartTime ≠ EndTime** → 400 `StartTimeEqualsEndTime`.
4. **Duration ≤ 16h** (M3) → 400 `DurationExceeds16Hours(17.5)`.
5. **Overlap detection** — không trùng ca khác trong cùng DayOfWeek. → 409 `OverlappingSchedule`.

`POST /api/v1/cafes/{cafeId}/staff-schedules/bulk` (H1):

1-5 như trên cho **từng** schedule.
6. **Atomic transaction** — nếu bất kỳ schedule nào fail validate, rollback toàn bộ (không có insert rác).

`POST /api/v1/cafes/{cafeId}/staff-schedules/copy-template` (C4):

1. Ownership (C5).
2. **Filter attendance** — chỉ copy staff có `ShiftAttendance` tuần nguồn. Fallback copy toàn bộ Active nếu tuần rỗng.
3. Atomic transaction.

`DELETE /api/v1/cafes/{cafeId}/staff/{staffId}` (C6):

1. Ownership.
2. Soft-cancel tất cả `StaffSchedule.Status = Cancelled` của staff trong cafe.
3. `StaffUnavailableDate` giữ nguyên (staff tự xóa nếu cần).
4. Xóa `CafeStaff` row.
5. Atomic transaction.

`DELETE /api/v1/cafes/{cafeId}/staff-schedules/{scheduleId}` (H7):

1. Ownership (IDOR-01).
2. **FK Restrict**: nếu có `ShiftAttendance` liên kết → 409 `ShiftAttendanceExistCannotDeleteSchedule`.
